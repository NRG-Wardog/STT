if(NOT WIN32)
    message(FATAL_ERROR "HebEN Dictation is Windows-only.")

file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/heben.ico" HEBEN_ICON_PATH)
configure_file(src/heben.rc.in "${CMAKE_CURRENT_BINARY_DIR}/heben.rc" @ONLY)
set_property(SOURCE "${CMAKE_CURRENT_BINARY_DIR}/heben.rc" APPEND PROPERTY OBJECT_DEPENDS "${HEBEN_ICON_PATH}")
add_executable(HebENDictation WIN32 src/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/heben.rc")
target_compile_definitions(HebENDictation PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(HebENDictation PRIVATE "${miniaudio_SOURCE_DIR}")
target_link_libraries(HebENDictation PRIVATE whisper user32 gdi32 shell32 ole32 oleaut32 uiautomationcore)
if(GGML_CUDA)
    find_package(CUDAToolkit REQUIRED)
    target_compile_definitions(HebENDictation PRIVATE HEBEN_CUDA_BUILD=1)
    target_link_libraries(HebENDictation PRIVATE CUDA::cudart)
    if(MSVC)
        # Keep cuBLAS and its large dependent DLL out of the idle process.
        target_compile_definitions(HebENDictation PRIVATE "HEBEN_CUBLAS_DLL=\"cublas64_${CUDAToolkit_VERSION_MAJOR}.dll\"")
        target_link_options(HebENDictation PRIVATE "/DELAYLOAD:cublas64_${CUDAToolkit_VERSION_MAJOR}.dll" "/DELAY:UNLOAD")
        target_link_libraries(HebENDictation PRIVATE delayimp)
    endif()
endif()
if(MSVC)
    target_compile_options(HebENDictation PRIVATE /W4 /EHsc /utf-8)
endif()
